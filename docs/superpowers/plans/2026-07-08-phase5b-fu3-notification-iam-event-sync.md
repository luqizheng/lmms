# Phase 5B-fu3 Implementation Plan — notification → iam 事件化

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** notification-service 移除同步 HTTP `by-{role,dept}` 调用，改 RabbitMQ 事件驱动本地 user membership 投影（user → role_ids, dept_id），单事务内 6 处 emit 点覆盖全 iam 写路径，bootstrap 端点一次性全量同步。

**Architecture:** iam-service 新增 3 类 membership 事件（`iam.user.role.{assigned,unassigned}` + `iam.user.dept.changed`），tx-in outbox emit；notification 启动一次 `GET /iam/internal/membership-bootstrap` 全量同步后，consumer 增量维护 `ntf_user_projection` 表；`publish_announcement` 改走本地 SQL `JSON_CONTAINS` 查询 role / dept 范围。HTTP 端点保留 deprecated。

**Tech Stack:** Rust 1.96 (workspace) · sqlx 0.8 (MySQL/JSON) · common-event (existing) · RabbitMQ lapin · wiremock (test) · testcontainers (integration)

//! Membership event emit helpers (Phase 5B-fu3).
//!
//! Tx-in outbox emit following the `enqueue_permission_changed` pattern.
//! All helpers take `&mut Transaction` so the emit is atomic with the
//! caller's SQL mutation.

use common_event::events::{
    IamUserDeptChanged, IamUserRoleAssigned, IamUserRoleUnassigned,
};
use common_event::{Envelope, MySqlOutboxRepo, OutboxRepo};
use sqlx::{MySql, Transaction};

use crate::error::IamError;

pub async fn enqueue_role_assigned(
    tx: &mut Transaction<'_, MySql>,
    repo: &MySqlOutboxRepo,
    user_id: &str,
    role_id: &str,
    tenant_id: i64,
    actor_id: Option<&str>,
    ts: i64,
) -> Result<(), IamError> {
    let mut env = Envelope::new(
        "iam-service",
        IamUserRoleAssigned {
            user_id: user_id.to_string(),
            role_id: role_id.to_string(),
            tenant_id,
            actor_id: actor_id.map(String::from),
            ts,
        },
    )
    .with_tenant_id(tenant_id.to_string());
    repo.enqueue(tx, &mut env)
        .await
        .map_err(|e| IamError::Internal(e.to_string()))

```rust
use std::collections::HashMap;
use axum::{extract::State, Extension, Json};
use common_db::WithTenantScope;
use jwt_ctx::Principal;
use serde::Serialize;
use sqlx::Row;
use crate::state::AppState;
use crate::error::IamError;

#[derive(Serialize)]
pub struct MembershipBootstrap {
    pub tenant_id: i64,
    pub mappings: HashMap<String, UserMembership>,
}

#[derive(Serialize)]
pub struct UserMembership {
    pub role_ids: Vec<String>,
    pub dept_id: Option<String>,
}

pub async fn membership_bootstrap(
    State(state): State<AppState>,
    Extension(principal): Extension<Principal>,
) -> Result<Json<MembershipBootstrap>, IamError> {
    let tenant_id = principal.tenant_id().unwrap_or(1);

    let rows = sqlx::query(
        "SELECT u.id, u.org_unit_id, GROUP_CONCAT(ur.role_id) AS role_ids \
         FROM iam_users u \
         LEFT JOIN iam_user_roles ur ON ur.user_id = u.id \
         WHERE u.tenant_id = ? AND u.deleted_at IS NULL \
         GROUP BY u.id, u.org_unit_id"
    )
    .bind(tenant_id)
    .fetch_all(&state.pool)
    .await
    .map_err(|e| IamError::Internal(e.to_string()))?;

    let mut mappings = HashMap::new();
    for row in rows {
        let user_id: String = row.try_get("id").map_err(|e| IamError::Internal(e.to_string()))?;
        let dept_id: Option<String> = row.try_get("org_unit_id").ok();
        let role_ids_str: Option<String> = row.try_get("role_ids").ok();
        let role_ids: Vec<String> = role_ids_str
            .map(|s| s.split(',').map(String::from).collect())
            .unwrap_or_default();
        mappings.insert(user_id, UserMembership { role_ids, dept_id });
    }

    Ok(Json(MembershipBootstrap { tenant_id, mappings }))
}
```

```rust
pub async fn apply_role_assigned(
    &self, user_id: &str, role_id: &str, tenant_id: i64, ts: i64,
) -> AppResult<()> {
    let mut tx = self.pool.begin().await?;
    let row: Option<(i64,)> = sqlx::query_as(
        "SELECT last_event_at FROM ntf_user_projection WHERE user_id = ? FOR UPDATE"
    ).bind(user_id).fetch_optional(&mut *tx).await?;
    let current_ts = row.map(|(t,)| t).unwrap_or(0);
    if ts < current_ts { return Ok(()); } // out-of-order drop

    if row.is_some() {
        sqlx::query(
            "UPDATE ntf_user_projection SET role_ids = JSON_ARRAY_APPEND(role_ids, '$', ?), last_event_at = ? WHERE user_id = ?"
        ).bind(role_id).bind(ts).bind(user_id).execute(&mut *tx).await?;
    } else {
        sqlx::query(
            "INSERT INTO ntf_user_projection (user_id, tenant_id, role_ids, dept_id, last_event_at) VALUES (?,?,JSON_ARRAY(?),?,?)"
        ).bind(user_id).bind(tenant_id).bind(role_id).bind(Option::<String>::None).bind(ts).execute(&mut *tx).await?;
    }
    tx.commit().await?;
    Ok(())
}
```

pub async fn resolve_users_by_scopes(
    &self, scopes: &[(ScopeType, String)], scope: &TenantScope,
) -> AppResult<HashSet<String>> {
    let tenant_id = scope.tenant_id().unwrap_or(1);
    let mut out = HashSet::new();
    for (t, id) in scopes {
        match t {
            ScopeType::User => { out.insert(id.clone()); }
            ScopeType::Role => {
                let rows: Vec<(String,)> = sqlx::query_as(
                    "SELECT user_id FROM ntf_user_projection \
                     WHERE tenant_id = ? AND JSON_CONTAINS(role_ids, JSON_QUOTE(?))",
                ).bind(tenant_id).bind(id).fetch_all(&self.pool).await?;
                out.extend(rows.into_iter().map(|(s,)| s));
            }
            ScopeType::Dept => {
                let rows: Vec<(String,)> = sqlx::query_as(
                    "SELECT user_id FROM ntf_user_projection \
                     WHERE tenant_id = ? AND dept_id = ?",
                ).bind(tenant_id).bind(id).fetch_all(&self.pool).await?;
                out.extend(rows.into_iter().map(|(s,)| s));
            }
        }
    }
    Ok(out)
}
```

    &self, tenant_id: i64, mappings: &HashMap<String, UserMembership>,

//! services/notification-service/src/consumer/membership.rs
//!
//! Durable consumer for `notification.membership-events` queue.
//! Routing keys: iam.user.role.assigned, iam.user.role.unassigned, iam.user.dept.changed.
//! Idempotent: apply by (user_id, ts) ordering, out-of-order events dropped.

use crate::state::AppState;
use common_event::subscribe_with_handler;
use common_event::HandlerError;
use serde_json::Value;
use tracing::{warn, error};

const QUEUE_NAME: &str = "notification.membership-events";
const ROUTING_KEYS: &[&str] = &[
    "iam.user.role.assigned",
    "iam.user.role.unassigned",
    "iam.user.dept.changed",
];

pub async fn run(state: AppState, mut shutdown: tokio::sync::watch::Receiver<bool>) -> anyhow::Result<()> {
    subscribe_with_handler(
        &state.event_bus,
        QUEUE_NAME,
        ROUTING_KEYS,
        |envelope: Value| async move {
            apply_membership_event(&state, envelope).await
        },
        shutdown,
    ).await
}

async fn apply_membership_event(state: &AppState, envelope: Value) -> Result<(), HandlerError> {
    let routing_key = envelope["routing_key"].as_str().unwrap_or("");
    let payload = &envelope["payload"];
    let user_id = payload["user_id"].as_str().ok_or_else(|| HandlerError::Permanent("missing user_id".into()))?;
    let tenant_id = payload["tenant_id"].as_i64().ok_or_else(|| HandlerError::Permanent("missing tenant_id".into()))?;
    let ts = payload["ts"].as_i64().unwrap_or(0);

    let result = match routing_key {
        "iam.user.role.assigned" => {
            let role_id = payload["role_id"].as_str().ok_or_else(|| HandlerError::Permanent("missing role_id".into()))?;
            state.db.apply_role_assigned(user_id, role_id, tenant_id, ts).await
        }
        "iam.user.role.unassigned" => {
            let role_id = payload["role_id"].as_str().ok_or_else(|| HandlerError::Permanent("missing role_id".into()))?;
            state.db.apply_role_unassigned(user_id, role_id, tenant_id, ts).await
        }
        "iam.user.dept.changed" => {
            let dept_id = payload["dept_id"].as_str();
            state.db.apply_dept_changed(user_id, dept_id, tenant_id, ts).await
        }
        _ => {
            warn!(routing_key, "unknown membership event routing_key");
            return Ok(());
        }
    };
    result.map_err(|e| {
        error!(error=%e, routing_key, user_id, "membership apply failed");
        HandlerError::Transient(e.to_string())
    })
}
```
